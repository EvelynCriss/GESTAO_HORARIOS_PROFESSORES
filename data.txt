CREATE DATABASE `sistema_horario_professores`


USE sistema_horario_professoreshorario_professores;


-- TABELA PROFESSORES

CREATE TABLE professores (

    id INT AUTO_INCREMENT PRIMARY KEY,

    nome VARCHAR(100) NOT NULL,

    email VARCHAR(100)

);


-- TABELA CURSOS

CREATE TABLE cursos (

    id INT AUTO_INCREMENT PRIMARY KEY,

    nome VARCHAR(100) NOT NULL,

    periodo VARCHAR(20),

    campus VARCHAR(100)

);


-- TABELA DISCIPLINAS

CREATE TABLE disciplinas (

    id INT AUTO_INCREMENT PRIMARY KEY,

    nome VARCHAR(100) NOT NULL,

    cor VARCHAR(20),

    curso_id INT,

    FOREIGN KEY (curso_id)
    REFERENCES cursos(id)

);


-- TABELA DISPONIBILIDADES

CREATE TABLE disponibilidades (

    id INT AUTO_INCREMENT PRIMARY KEY,

    professor_id INT,

    dia_semana ENUM(
        'Segunda',
        'Terça',
        'Quarta',
        'Quinta',
        'Sexta'
    ),

    FOREIGN KEY (professor_id)
    REFERENCES professores(id)

);


-- TABELA AULAS

CREATE TABLE aulas (

    id INT AUTO_INCREMENT PRIMARY KEY,

    professor_id INT,

    curso_id INT,

    disciplina_id INT,

    dia_semana ENUM(
        'Segunda',
        'Terça',
        'Quarta',
        'Quinta',
        'Sexta'
    ),

    FOREIGN KEY (professor_id)
    REFERENCES professores(id),

    FOREIGN KEY (curso_id)
    REFERENCES cursos(id),

    FOREIGN KEY (disciplina_id)
    REFERENCES disciplinas(id)

);
